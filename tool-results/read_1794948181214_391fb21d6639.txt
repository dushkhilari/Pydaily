    10→     CODE LAB — single-file Python playground
    11→     ------------------------------------------------------------------------
    12→     This page is built to be embedded in a notebook-themed learning site via
    13→     an <iframe>, but it also works perfectly when opened directly as a file.
    14→
    15→     HOW THE PIECES FIT TOGETHER
    16→       <head>   : Google Fonts + CodeMirror stylesheets + all page CSS (§1)
    17→       <body>   : slim top bar (status pill + Run/Clear buttons), the editor
    18→                  pane on TOP and the console pane BELOW it — stacked, so the
    19→                  editor gets the full width and lots of height on any screen
    20→       scripts  : pyodide.js (Python runtime) → CodeMirror JS (core FIRST,
    21→                  then the Python mode, then the addons) → app script (§2–§7)
    22→
    23→     EMBEDDING NOTES FOR THE HOST SITE
    24→       • Wait for the { type: "pyodide-ready" } postMessage before triggering
    25→         runs from outside the frame.
    26→       • Push code into the editor from the host page with:
    27→             iframe.contentWindow.postMessage(
    28→               { type: "set-code", code: "print('hi')" }, "*")
    29→       • If the host uses the `sandbox` attribute on the iframe, include
    30→         `allow-scripts` (required) and `allow-modals` (needed so that
    31→         Python's input() can open window.prompt()).

   483→========================================================================== -->
   484→<script>
   485→(function () {
   486→  "use strict";
   487→
   488→  /* ------------------------------------------------------------------
   489→     Shared DOM references & app state (used by every section below).
   490→     ------------------------------------------------------------------ */
   491→  var PYODIDE_INDEX_URL = "https://cdn.jsdelivr.net/pyodide/v0.26.4/full/";
   492→
   493→  var runBtn      = document.getElementById("run-btn");       // "Run Code" button
   494→  var clearBtn    = document.getElementById("clear-btn");     // "Clear Output" button
   495→  var editorHost  = document.getElementById("editor-host");   // div CodeMirror mounts into
   496→  var consoleBody = document.getElementById("console-body");  // ruled-paper output pane
   497→  var statusPill  = document.getElementById("status-pill");   // top-bar status pill
   498→  var statusText  = document.getElementById("status-text");   // pill text node
   499→  var bootOverlay = document.getElementById("boot-overlay");  // "Loading Python…" overlay
   500→
   501→  var pyodide   = null;   // Pyodide runtime instance (assigned during boot)
   502→  var pyReady   = false;  // true once loadPyodide() has resolved
   503→  var pyRunning = false;  // true while a user script is currently executing
   504→
   505→  // ================= 2. CODEMIRROR EDITOR SETUP =================
   506→  // Dark-themed editor with: Python syntax highlighting, line numbers,
   507→  // auto-indentation (indentUnit 4), bracket matching + auto-closing
   508→  // and active-line highlighting. Plain buttons only — no shortcuts.
   509→  var STARTER_CODE = `# ============================================
   510→#  Welcome to the Code Lab!
   511→#  Real Python 3 running in your browser
   512→#  (Pyodide + WebAssembly — no server needed)
   513→# ============================================
   514→
   515→name = "PyDaily"        # a string variable
   516→days = 100              # an integer variable
   517→
   518→print("Hello, World!")
   519→print(f"Welcome to {name} — {days} days of Python!")
   520→
   521→# a loop
   522→for day in range(1, 6):
   523→    print(f"Day {day}: keep the streak alive!")
   524→
   525→# some math
   526→import math
   527→
   528→print("The square root of 144 is", math.sqrt(144))
   529→print("The sum of 1..100 is", sum(range(1, 101)))
   530→
   531→# Bonus: this lesson uses input() via a browser popup.
   532→# Uncomment the two lines below and press Run!
   533→# your_name = input("What is your name? ")
   534→# print("Nice to meet you,", your_name)
   535→`;
   536→
   537→  var editor = CodeMirror(editorHost, {
   538→    value: STARTER_CODE,          // pre-filled "Hello, World!" script
   539→    mode: "python",               // syntax highlighting (mode loaded above)
   540→    theme: "material-darker",     // dark theme (CSS loaded above)
   541→    lineNumbers: true,
   542→    indentUnit: 4,                // Python-style 4-space indentation
   543→    matchBrackets: true,
   544→    autoCloseBrackets: true,
   545→    styleActiveLine: true
   546→  });
   547→
   548→  // ================= 3. PYODIDE PYTHON RUNTIME =================
   549→  // Boot sequence (runs on window load):
   550→  //   1. show the loading indicator + spinner pill, disable "Run Code"
   551→  //   2. await loadPyodide(...)  ← downloads CPython→WASM (a few seconds)
   552→  //   3. redirect stdout/stderr into the console pane (BEFORE user code runs)
   553→  //   4. install the input() → window.prompt() override (section 5)
   554→  //   5. mark ready: enable the button, flip the pill to "Python 3 · ready",
   555→  //      hide the overlay, and notify the host page (if embedded in an iframe)
   556→  // If the CDN fails, the catch block keeps the page alive and explains
   557→  // the problem in red inside the console pane.
   558→  function bootPyodide() {
   559→    bootOverlay.hidden = false;   // loading indicator over the console
   560→    runBtn.disabled = true;       // no running code until ready
   561→    setStatus("booting");         // pill: spinner + amber dot
   562→
   563→    try {
   564→      // Guard: pyodide.js may not have loaded (offline / blocked CDN).
   565→      if (typeof loadPyodide !== "function") {
   566→        throw new Error("loadPyodide() not found — the Pyodide CDN script did not load.");
   567→      }
   568→
   569→      loadPyodide({ indexURL: PYODIDE_INDEX_URL })
   570→        .then(function (py) {
   571→          pyodide = py;
   572→
   573→          // Redirect Python's stdout/stderr to the console pane.
   574→          // `batched` fires once per complete line of output.
   575→          pyodide.setStdout({ batched: function (s) { appendOutput(s + "\n", "out"); } });
   576→          pyodide.setStderr({ batched: function (s) { appendOutput(s + "\n", "err"); } });
   577→
   578→          // Python's input() opens a browser prompt (section 5).
   579→          installInputOverride();
   580→
   581→          pyReady = true;
   582→          bootOverlay.hidden = true;    // hide the loading indicator
   583→          runBtn.disabled = false;      // enable "Run Code"
   584→          setStatus("ready");           // pill: green "Python 3 · ready"
   585→          appendOutput("Python runtime ready — press Run Code to execute main.py.\n", "muted");
   586→
   587→          notifyHostReady();            // postMessage to the host page
   588→        })
   589→        .catch(function (err) {
   590→          bootOverlay.hidden = true;
   591→          setStatus("error");           // pill: red dot (CDN failure)
   592→          appendOutput("Could not load the Python runtime — check your internet connection.\n", "err");
   593→          console.error("[Code Lab] Failed to boot Pyodide:", err);
   594→        });
   595→    } catch (err) {
   596→      // Same failure path, for the synchronous typeof guard above.
   597→      bootOverlay.hidden = true;
   598→      setStatus("error");
   599→      appendOutput("Could not load the Python runtime — check your internet connection.\n", "err");
   600→      console.error("[Code Lab] Failed to boot Pyodide:", err);
   601→    }
   602→  }
   603→
   604→  // ================= 4. OUTPUT HANDLING (stdout/stderr) =================
   605→  // appendOutput(text, cls) is the single gateway for everything that
   606→  // appears in the console pane. Class names map to colors:
   607→  //   "out"   → default ink      (print output / stdout)
   608→  //   "err"   → red pen          (tracebacks / stderr)
   609→  //   "muted" → gray             (system messages)
   610→  // We use textContent (never innerHTML) so Python output can never
   611→  // inject HTML into the page, and we auto-scroll to the newest line.
   612→  function appendOutput(text, cls) {
   613→    var line = document.createElement("div");
   614→    line.className = cls || "out";
   615→    line.textContent = text;
   616→    consoleBody.appendChild(line);
   617→    consoleBody.scrollTop = consoleBody.scrollHeight; // auto-scroll
   618→  }
   619→
   620→  // clearOutput() empties the console pane (also called before each run).
   621→  function clearOutput() {
   622→    consoleBody.textContent = "";
   623→  }
   624→
   625→  // ================= 5. INPUT() VIA WINDOW.PROMPT =================
   626→  // Python's built-in input() blocks on stdin, which does not exist in a
   627→  // browser — so after boot we override it with a small Python shim that
   628→  // calls window.prompt() instead. Cancelling the prompt raises EOFError
   629→  // (shown as a red traceback), which is exactly what real Python does on
   630→  // an EOF. The override is wrapped in try/except-equivalent so a failure
   631→  // here can NEVER block the boot sequence.
   632→  var INPUT_OVERRIDE_PY = `
   633→import builtins, js
   634→
   635→def _input(prompt=""):
   636→    reply = js.window.prompt(str(prompt))
   637→    if reply is None:
   638→        raise EOFError("input() cancelled")
   639→    return str(reply)
   640→
   641→builtins.input = _input
   642→`;
   643→
   644→  function installInputOverride() {
   645→    try {
   646→      pyodide.runPython(INPUT_OVERRIDE_PY);
   647→    } catch (err) {
   648→      // input() simply keeps its default behavior if this fails.
   649→      console.warn("[Code Lab] Could not install input() override:", err);
   650→    }
   651→  }
   652→
   653→  // ================= 6. RUN / CLEAR / STATUS =================
   654→  // setStatus(state) drives the status pill through its states:
   655→  //   booting → "Booting Python runtime…" (spinner + amber dot)
   656→  //   ready   → "Python 3 · ready"        (green dot)
   657→  //   running → "Running…"                (blue pulsing dot)
   658→  //   error   → "Python failed to load"   (red dot, CDN failure only)
   659→  function setStatus(state) {
   660→    statusPill.classList.remove("booting", "ready", "running", "error");
   661→    statusPill.classList.add(state);
   662→    statusText.textContent = {
   663→      booting: "Booting Python runtime…",
   664→      ready:   "Python 3 · ready",
   665→      running: "Running…",
   666→      error:   "Python failed to load"
   667→    }[state];
   668→  }
   669→
   670→  // runCode() executes the editor's Python code:
   671→  //   • ignored when the runtime is not ready or a run is already active
   672→  //   • clears the console, prints a muted "▸ run started"
   673→  //   • awaits pyodide.runPythonAsync(code); any Python traceback arrives
   674→  //     as err.message and is printed in RED without crashing the page
   675→  //   • a `finally` block always restores the UI state (muted "▸ run
   676→  //     finished", button re-enabled, pill back to "Python 3 · ready")
   677→  function runCode() {
   678→    if (!pyReady || pyRunning || !pyodide) { return; } // do nothing if not ready/busy
   679→    pyRunning = true;
   680→    runBtn.disabled = true;
   681→    setStatus("running");
   682→
   683→    clearOutput();                          // fresh console for each run
   684→    appendOutput("\u25B8 run started\n", "muted");
   685→
   686→    var code = editor.getValue();
   687→    pyodide.runPythonAsync(code)
   688→      .catch(function (err) {
   689→        appendOutput((err && err.message ? err.message : String(err)) + "\n", "err");
   690→      })
   691→      .then(function () {
   692→        // `finally` equivalent: always runs, success or failure.
   693→        pyRunning = false;
   694→        appendOutput("\u25B8 run finished\n", "muted");
   695→        if (pyReady) {
   696→          runBtn.disabled = false;
   697→          setStatus("ready");
   698→        }
   699→      });
   700→  }
   701→
   702→  // Wire up the toolbar buttons.
   703→  runBtn.addEventListener("click", runCode);
   704→  clearBtn.addEventListener("click", clearOutput);
   705→
   706→  // ================= 7. CROSS-FRAME API (set-code) =================
   707→  // When embedded in an iframe, the host notebook site can push code into
   708→  // the editor by posting a message:
   709→  //     iframe.contentWindow.postMessage({ type: "set-code", code: "..." }, "*")
   710→  // This listener is harmless when the page is opened standalone (no
   711→  // messages ever arrive).
   712→  window.addEventListener("message", function (event) {
   713→    if (event.data && event.data.type === "set-code") {
   714→      editor.setValue(String(event.data.code ?? ""));
   715→    }
   716→  });
   717→
   718→  // Once Pyodide is ready, tell the host page we can accept code / runs.
   719→  // When standalone, window.parent is this same window — posting to
   720→  // ourselves is harmless (the listener above only reacts to "set-code").
   721→  function notifyHostReady() {
   722→    try {
   723→      window.parent.postMessage({ type: "pyodide-ready" }, "*");
   724→    } catch (err) { /* ignore — standalone page or restricted sandbox */ }
   725→  }
   726→
   727→  /* ------------------------------------------------------------------
   728→     Kick off the boot sequence when the page has fully loaded.
   729→     ------------------------------------------------------------------ */
   730→  if (document.readyState === "complete") {
   731→    bootPyodide();
   732→  } else {
   733→    window.addEventListener("load", bootPyodide);
   734→  }
   735→})();
   736→</script>
   737→
   738→</body>
