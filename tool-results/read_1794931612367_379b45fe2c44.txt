    10→     CODE LAB — single-file Python playground
    11→     ------------------------------------------------------------------------
    12→     This page is built to be embedded in a notebook-themed learning site via
    13→     an <iframe>, but it also works perfectly when opened directly as a file.
    14→
    15→     HOW THE PIECES FIT TOGETHER
    16→       <head>   : Google Fonts + CodeMirror stylesheets + all page CSS (§1)
    17→       <body>   : header / toolbar / editor + console grid / footer markup
    18→       scripts  : pyodide.js (Python runtime) → CodeMirror JS (core FIRST,
    19→                  then the Python mode, then the addons) → app script (§2–§7)
    20→
    21→     EMBEDDING NOTES FOR THE HOST SITE
    22→       • Wait for the { type: "pyodide-ready" } postMessage before triggering
    23→         runs from outside the frame.
    24→       • Push code into the editor from the host page with:
    25→             iframe.contentWindow.postMessage(
    26→               { type: "set-code", code: "print('hi')" }, "*")
    27→       • If the host uses the `sandbox` attribute on the iframe, include
    28→         `allow-scripts` (required) and `allow-modals` (needed so that
    29→         Python's input() can open window.prompt()).

   485→<!-- ==========================================================================
   486→     APPLICATION SCRIPT — everything the playground does lives here.
   487→========================================================================== -->
   488→<script>
   489→(function () {
   490→  "use strict";
   491→
   492→  /* ------------------------------------------------------------------
   493→     Shared DOM references & app state (used by every section below).
   494→     ------------------------------------------------------------------ */
   495→  var PYODIDE_INDEX_URL = "https://cdn.jsdelivr.net/pyodide/v0.26.4/full/";
   496→
   497→  var runBtn      = document.getElementById("run-btn");       // "Run Code" button
   498→  var clearBtn    = document.getElementById("clear-btn");     // "Clear Output" button
   499→  var editorHost  = document.getElementById("editor-host");   // div CodeMirror mounts into
   500→  var consoleBody = document.getElementById("console-body");  // ruled-paper output pane
   501→  var statusPill  = document.getElementById("status-pill");   // header status pill
   502→  var statusText  = document.getElementById("status-text");   // pill text node
   503→  var bootOverlay = document.getElementById("boot-overlay");  // "Loading Python…" overlay
   504→
   505→  var pyodide   = null;   // Pyodide runtime instance (assigned during boot)
   506→  var pyReady   = false;  // true once loadPyodide() has resolved
   507→  var pyRunning = false;  // true while a user script is currently executing
   508→
   509→  // ================= 2. CODEMIRROR EDITOR SETUP =================
   510→  // Dark-themed editor with: Python syntax highlighting, line numbers,
   511→  // auto-indentation (indentUnit 4), bracket matching + auto-closing,
   512→  // active-line highlighting and Ctrl/Cmd+Enter to run.
   513→  // NOTE: `runCode` (section 6) is a hoisted *function declaration*, so
   514→  // referencing it in `extraKeys` here is safe even though its body is
   515→  // written later in this file.
   516→  var STARTER_CODE = `# ============================================
   517→#  Welcome to the Code Lab!
   518→#  Real Python 3 running in your browser
   519→#  (Pyodide + WebAssembly — no server needed)
   520→# ============================================
   521→
   522→name = "PyDaily"        # a string variable
   523→days = 100              # an integer variable
   524→
   525→print("Hello, World!")
   526→print(f"Welcome to {name} — {days} days of Python!")
   527→
   528→# a loop
   529→for day in range(1, 6):
   530→    print(f"Day {day}: keep the streak alive!")
   531→
   532→# some math
   533→import math
   534→
   535→print("The square root of 144 is", math.sqrt(144))
   536→print("The sum of 1..100 is", sum(range(1, 101)))
   537→
   538→# Bonus: this lesson uses input() via a browser popup.
   539→# Uncomment the two lines below and press Run!
   540→# your_name = input("What is your name? ")
   541→# print("Nice to meet you,", your_name)
   542→`;
   543→
   544→  var editor = CodeMirror(editorHost, {
   545→    value: STARTER_CODE,          // pre-filled "Hello, World!" script
   546→    mode: "python",               // syntax highlighting (mode loaded above)
   547→    theme: "material-darker",     // dark theme (CSS loaded above)
   548→    lineNumbers: true,
   549→    indentUnit: 4,                // Python-style 4-space indentation
   550→    matchBrackets: true,
   551→    autoCloseBrackets: true,
   552→    styleActiveLine: true,
   553→    extraKeys: { "Ctrl-Enter": runCode, "Cmd-Enter": runCode }
   554→  });
   555→
   556→  // ================= 3. PYODIDE PYTHON RUNTIME =================
   557→  // Boot sequence (runs on window load):
   558→  //   1. show the loading indicator + spinner pill, disable "Run Code"
   559→  //   2. await loadPyodide(...)  ← downloads CPython→WASM (a few seconds)
   560→  //   3. redirect stdout/stderr into the console pane (BEFORE user code runs)
   561→  //   4. install the input() → window.prompt() override (section 5)
   562→  //   5. mark ready: enable the button, flip the pill to "Python 3 · ready",
   563→  //      hide the overlay, and notify the host page (if embedded in an iframe)
   564→  // If the CDN fails, the catch block keeps the page alive and explains
   565→  // the problem in red inside the console pane.
   566→  function bootPyodide() {
   567→    bootOverlay.hidden = false;   // loading indicator over the console
   568→    runBtn.disabled = true;       // no running code until ready
   569→    setStatus("booting");         // pill: spinner + amber dot
   570→
   571→    try {
   572→      // Guard: pyodide.js may not have loaded (offline / blocked CDN).
   573→      if (typeof loadPyodide !== "function") {
   574→        throw new Error("loadPyodide() not found — the Pyodide CDN script did not load.");
   575→      }
   576→
   577→      loadPyodide({ indexURL: PYODIDE_INDEX_URL })
   578→        .then(function (py) {
   579→          pyodide = py;
   580→
   581→          // Redirect Python's stdout/stderr to the console pane.
   582→          // `batched` fires once per complete line of output.
   583→          pyodide.setStdout({ batched: function (s) { appendOutput(s + "\n", "out"); } });
   584→          pyodide.setStderr({ batched: function (s) { appendOutput(s + "\n", "err"); } });
   585→
   586→          // Python's input() opens a browser prompt (section 5).
   587→          installInputOverride();
   588→
   589→          pyReady = true;
   590→          bootOverlay.hidden = true;    // hide the loading indicator
   591→          runBtn.disabled = false;      // enable "Run Code"
   592→          setStatus("ready");           // pill: green "Python 3 · ready"
   593→          appendOutput("Python runtime ready — press Run Code (or Ctrl+Enter) to execute main.py.\n", "muted");
   594→
   595→          notifyHostReady();            // postMessage to the host page
   596→        })
   597→        .catch(function (err) {
   598→          bootOverlay.hidden = true;
   599→          setStatus("error");           // pill: red dot (CDN failure)
   600→          appendOutput("Could not load the Python runtime — check your internet connection.\n", "err");
   601→          console.error("[Code Lab] Failed to boot Pyodide:", err);
   602→        });
   603→    } catch (err) {
   604→      // Same failure path, for the synchronous typeof guard above.
   605→      bootOverlay.hidden = true;
   606→      setStatus("error");
   607→      appendOutput("Could not load the Python runtime — check your internet connection.\n", "err");
   608→      console.error("[Code Lab] Failed to boot Pyodide:", err);
   609→    }
   610→  }
   611→
   612→  // ================= 4. OUTPUT HANDLING (stdout/stderr) =================
   613→  // appendOutput(text, cls) is the single gateway for everything that
   614→  // appears in the console pane. Class names map to colors:
   615→  //   "out"   → default ink      (print output / stdout)
   616→  //   "err"   → red pen          (tracebacks / stderr)
   617→  //   "muted" → gray             (system messages)
   618→  // We use textContent (never innerHTML) so Python output can never
   619→  // inject HTML into the page, and we auto-scroll to the newest line.
   620→  function appendOutput(text, cls) {
   621→    var line = document.createElement("div");
   622→    line.className = cls || "out";
   623→    line.textContent = text;
   624→    consoleBody.appendChild(line);
   625→    consoleBody.scrollTop = consoleBody.scrollHeight; // auto-scroll
   626→  }
   627→
   628→  // clearOutput() empties the console pane (also called before each run).
   629→  function clearOutput() {
   630→    consoleBody.textContent = "";
   631→  }
   632→
   633→  // ================= 5. INPUT() VIA WINDOW.PROMPT =================
   634→  // Python's built-in input() blocks on stdin, which does not exist in a
   635→  // browser — so after boot we override it with a small Python shim that
   636→  // calls window.prompt() instead. Cancelling the prompt raises EOFError
   637→  // (shown as a red traceback), which is exactly what real Python does on
   638→  // an EOF. The override is wrapped in try/except-equivalent so a failure
   639→  // here can NEVER block the boot sequence.
   640→  var INPUT_OVERRIDE_PY = `
   641→import builtins, js
   642→
   643→def _input(prompt=""):
   644→    reply = js.window.prompt(str(prompt))
   645→    if reply is None:
   646→        raise EOFError("input() cancelled")
   647→    return str(reply)
   648→
   649→builtins.input = _input
   650→`;
   651→
   652→  function installInputOverride() {
   653→    try {
   654→      pyodide.runPython(INPUT_OVERRIDE_PY);
   655→    } catch (err) {
   656→      // input() simply keeps its default behavior if this fails.
   657→      console.warn("[Code Lab] Could not install input() override:", err);
   658→    }
   659→  }
   660→
   661→  // ================= 6. RUN / CLEAR / STATUS =================
   662→  // setStatus(state) drives the header status pill through its states:
   663→  //   booting → "Booting Python runtime…" (spinner + amber dot)
   664→  //   ready   → "Python 3 · ready"        (green dot)
   665→  //   running → "Running…"                (blue pulsing dot)
   666→  //   error   → "Python failed to load"   (red dot, CDN failure only)
   667→  function setStatus(state) {
   668→    statusPill.classList.remove("booting", "ready", "running", "error");
   669→    statusPill.classList.add(state);
   670→    statusText.textContent = {
   671→      booting: "Booting Python runtime…",
   672→      ready:   "Python 3 · ready",
   673→      running: "Running…",
   674→      error:   "Python failed to load"
   675→    }[state];
   676→  }
   677→
   678→  // runCode() executes the editor's Python code:
   679→  //   • ignored when the runtime is not ready or a run is already active
   680→  //   • clears the console, prints a muted "▸ run started"
   681→  //   • awaits pyodide.runPythonAsync(code); any Python traceback arrives
   682→  //     as err.message and is printed in RED without crashing the page
   683→  //   • a `finally` block always restores the UI state (muted "▸ run
   684→  //     finished", button re-enabled, pill back to "Python 3 · ready")
   685→  function runCode() {
   686→    if (!pyReady || pyRunning || !pyodide) { return; } // do nothing if not ready/busy
   687→    pyRunning = true;
   688→    runBtn.disabled = true;
   689→    setStatus("running");
   690→
   691→    clearOutput();                          // fresh console for each run
   692→    appendOutput("\u25B8 run started\n", "muted");
   693→
   694→    var code = editor.getValue();
   695→    pyodide.runPythonAsync(code)
   696→      .catch(function (err) {
   697→        appendOutput((err && err.message ? err.message : String(err)) + "\n", "err");
   698→      })
   699→      .then(function () {
   700→        // `finally` equivalent: always runs, success or failure.
   701→        pyRunning = false;
   702→        appendOutput("\u25B8 run finished\n", "muted");
   703→        if (pyReady) {
   704→          runBtn.disabled = false;
   705→          setStatus("ready");
   706→        }
   707→      });
   708→  }
   709→
   710→  // Wire up the toolbar buttons.
   711→  runBtn.addEventListener("click", runCode);
   712→  clearBtn.addEventListener("click", clearOutput);
   713→
   714→  // ================= 7. CROSS-FRAME API (set-code) =================
   715→  // When embedded in an iframe, the host notebook site can push code into
   716→  // the editor by posting a message:
   717→  //     iframe.contentWindow.postMessage({ type: "set-code", code: "..." }, "*")
   718→  // This listener is harmless when the page is opened standalone (no
   719→  // messages ever arrive).
   720→  window.addEventListener("message", function (event) {
   721→    if (event.data && event.data.type === "set-code") {
   722→      editor.setValue(String(event.data.code ?? ""));
   723→    }
   724→  });
   725→
   726→  // Once Pyodide is ready, tell the host page we can accept code / runs.
   727→  // When standalone, window.parent is this same window — posting to
   728→  // ourselves is harmless (the listener above only reacts to "set-code").
   729→  function notifyHostReady() {
   730→    try {
   731→      window.parent.postMessage({ type: "pyodide-ready" }, "*");
   732→    } catch (err) { /* ignore — standalone page or restricted sandbox */ }
   733→  }
   734→
   735→  /* ------------------------------------------------------------------
   736→     Kick off the boot sequence when the page has fully loaded.
   737→     ------------------------------------------------------------------ */
   738→  if (document.readyState === "complete") {
   739→    bootPyodide();
   740→  } else {
   741→    window.addEventListener("load", bootPyodide);
   742→  }
   743→})();
