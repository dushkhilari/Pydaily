    11→    11→     ------------------------------------------------------------------------
    12→    12→     This page is built to be embedded in a notebook-themed learning site via
    13→    13→     an <iframe>, but it also works perfectly when opened directly as a file.
    14→    14→
    15→    15→     HOW THE PIECES FIT TOGETHER
    16→    16→       <head>   : Google Fonts + CodeMirror stylesheets + all page CSS (§1)
    17→    17→       <body>   : slim top bar (status pill + Run/Clear buttons), the editor
    18→    18→                  pane on TOP and the console pane BELOW it — stacked, so the
    19→    19→                  editor gets the full width and lots of height on any screen
    20→    20→       scripts  : pyodide.js (Python runtime) → CodeMirror JS (core FIRST,
    21→    21→                  then the Python mode, then the addons) → app script (§2–§7)
    22→    22→
    23→    23→     EMBEDDING NOTES FOR THE HOST SITE
    24→    24→       • Wait for the { type: "pyodide-ready" } postMessage before triggering
    25→    25→         runs from outside the frame.
    26→    26→       • Push code into the editor from the host page with:
    27→    27→             iframe.contentWindow.postMessage(
    28→    28→               { type: "set-code", code: "print('hi')" }, "*")
    29→    29→       • If the host uses the `sandbox` attribute on the iframe, include
    30→    30→         `allow-scripts` (required) and `allow-modals` (needed so that
    31→    31→         Python's input() can open window.prompt()).

   485→   485→(function () {
   486→   486→  "use strict";
   487→   487→
   488→   488→  /* ------------------------------------------------------------------
   489→   489→     Shared DOM references & app state (used by every section below).
   490→   490→     ------------------------------------------------------------------ */
   491→   491→  var PYODIDE_INDEX_URL = "https://cdn.jsdelivr.net/pyodide/v0.26.4/full/";
   492→   492→
   493→   493→  var runBtn      = document.getElementById("run-btn");       // "Run Code" button
   494→   494→  var clearBtn    = document.getElementById("clear-btn");     // "Clear Output" button
   495→   495→  var editorHost  = document.getElementById("editor-host");   // div CodeMirror mounts into
   496→   496→  var consoleBody = document.getElementById("console-body");  // ruled-paper output pane
   497→   497→  var statusPill  = document.getElementById("status-pill");   // top-bar status pill
   498→   498→  var statusText  = document.getElementById("status-text");   // pill text node
   499→   499→  var bootOverlay = document.getElementById("boot-overlay");  // "Loading Python…" overlay
   500→   500→
   501→   501→  var pyodide   = null;   // Pyodide runtime instance (assigned during boot)
   502→   502→  var pyReady   = false;  // true once loadPyodide() has resolved
   503→   503→  var pyRunning = false;  // true while a user script is currently executing
   504→   504→
   505→   505→  // ================= 2. CODEMIRROR EDITOR SETUP =================
   506→   506→  // Dark-themed editor with: Python syntax highlighting, line numbers,
   507→   507→  // auto-indentation (indentUnit 4), bracket matching + auto-closing
   508→   508→  // and active-line highlighting. Plain buttons only — no shortcuts.
   509→   509→  var STARTER_CODE = `# ============================================
   510→   510→#  Welcome to the Code Lab!
   511→   511→#  Real Python 3 running in your browser
   512→   512→#  (Pyodide + WebAssembly — no server needed)
   513→   513→# ============================================
   514→   514→
   515→   515→name = "PyDaily"        # a string variable
   516→   516→days = 100              # an integer variable
   517→   517→
   518→   518→print("Hello, World!")
   519→   519→print(f"Welcome to {name} — {days} days of Python!")
   520→   520→
   521→   521→# a loop
   522→   522→for day in range(1, 6):
   523→   523→    print(f"Day {day}: keep the streak alive!")
   524→   524→
   525→   525→# some math
   526→   526→import math
   527→   527→
   528→   528→print("The square root of 144 is", math.sqrt(144))
   529→   529→print("The sum of 1..100 is", sum(range(1, 101)))
   530→   530→
   531→   531→# Bonus: this lesson uses input() via a browser popup.
   532→   532→# Uncomment the two lines below and press Run!
   533→   533→# your_name = input("What is your name? ")
   534→   534→# print("Nice to meet you,", your_name)
   535→   535→`;
   536→   536→
   537→   537→  var editor = CodeMirror(editorHost, {
   538→   538→    value: STARTER_CODE,          // pre-filled "Hello, World!" script
   539→   539→    mode: "python",               // syntax highlighting (mode loaded above)
   540→   540→    theme: "material-darker",     // dark theme (CSS loaded above)
   541→   541→    lineNumbers: true,
   542→   542→    indentUnit: 4,                // Python-style 4-space indentation
   543→   543→    matchBrackets: true,
   544→   544→    autoCloseBrackets: true,
   545→   545→    styleActiveLine: true
   546→   546→  });
   547→   547→
   548→   548→  // ================= 3. PYODIDE PYTHON RUNTIME =================
   549→   549→  // Boot sequence (runs on window load):
   550→   550→  //   1. show the loading indicator + spinner pill, disable "Run Code"
   551→   551→  //   2. await loadPyodide(...)  ← downloads CPython→WASM (a few seconds)
   552→   552→  //   3. redirect stdout/stderr into the console pane (BEFORE user code runs)
   553→   553→  //   4. install the input() → window.prompt() override (section 5)
   554→   554→  //   5. mark ready: enable the button, flip the pill to "Python 3 · ready",
   555→   555→  //      hide the overlay, and notify the host page (if embedded in an iframe)
   556→   556→  // If the CDN fails, the catch block keeps the page alive and explains
   557→   557→  // the problem in red inside the console pane.
   558→   558→  function bootPyodide() {
   559→   559→    bootOverlay.hidden = false;   // loading indicator over the console
   560→   560→    runBtn.disabled = true;       // no running code until ready
   561→   561→    setStatus("booting");         // pill: spinner + amber dot
   562→   562→
   563→   563→    try {
   564→   564→      // Guard: pyodide.js may not have loaded (offline / blocked CDN).
   565→   565→      if (typeof loadPyodide !== "function") {
   566→   566→        throw new Error("loadPyodide() not found — the Pyodide CDN script did not load.");
   567→   567→      }
   568→   568→
   569→   569→      loadPyodide({ indexURL: PYODIDE_INDEX_URL })
   570→   570→        .then(function (py) {
   571→   571→          pyodide = py;
   572→   572→
   573→   573→          // Redirect Python's stdout/stderr to the console pane.
   574→   574→          // `batched` fires once per complete line of output.
   575→   575→          pyodide.setStdout({ batched: function (s) { appendOutput(s + "\n", "out"); } });
   576→   576→          pyodide.setStderr({ batched: function (s) { appendOutput(s + "\n", "err"); } });
   577→   577→
   578→   578→          // Python's input() opens a browser prompt (section 5).
   579→   579→          installInputOverride();
   580→   580→
   581→   581→          pyReady = true;
   582→   582→          bootOverlay.hidden = true;    // hide the loading indicator
   583→   583→          runBtn.disabled = false;      // enable "Run Code"
   584→   584→          setStatus("ready");           // pill: green "Python 3 · ready"
   585→   585→          appendOutput("Python runtime ready — press Run Code to execute main.py.\n", "muted");
   586→   586→
   587→   587→          notifyHostReady();            // postMessage to the host page
   588→   588→        })
   589→   589→        .catch(function (err) {
   590→   590→          bootOverlay.hidden = true;
   591→   591→          setStatus("error");           // pill: red dot (CDN failure)
   592→   592→          appendOutput("Could not load the Python runtime — check your internet connection.\n", "err");
   593→   593→          console.error("[Code Lab] Failed to boot Pyodide:", err);
   594→   594→        });
   595→   595→    } catch (err) {
   596→   596→      // Same failure path, for the synchronous typeof guard above.
   597→   597→      bootOverlay.hidden = true;
   598→   598→      setStatus("error");
   599→   599→      appendOutput("Could not load the Python runtime — check your internet connection.\n", "err");
   600→   600→      console.error("[Code Lab] Failed to boot Pyodide:", err);
   601→   601→    }
   602→   602→  }
   603→   603→
   604→   604→  // ================= 4. OUTPUT HANDLING (stdout/stderr) =================
   605→   605→  // appendOutput(text, cls) is the single gateway for everything that
   606→   606→  // appears in the console pane. Class names map to colors:
   607→   607→  //   "out"   → default ink      (print output / stdout)
   608→   608→  //   "err"   → red pen          (tracebacks / stderr)
   609→   609→  //   "muted" → gray             (system messages)
   610→   610→  // We use textContent (never innerHTML) so Python output can never
   611→   611→  // inject HTML into the page, and we auto-scroll to the newest line.
   612→   612→  function appendOutput(text, cls) {
   613→   613→    var line = document.createElement("div");
   614→   614→    line.className = cls || "out";
   615→   615→    line.textContent = text;
   616→   616→    consoleBody.appendChild(line);
   617→   617→    consoleBody.scrollTop = consoleBody.scrollHeight; // auto-scroll
   618→   618→  }
   619→   619→
   620→   620→  // clearOutput() empties the console pane (also called before each run).
   621→   621→  function clearOutput() {
   622→   622→    consoleBody.textContent = "";
   623→   623→  }
   624→   624→
   625→   625→  // ================= 5. INPUT() VIA WINDOW.PROMPT =================
   626→   626→  // Python's built-in input() blocks on stdin, which does not exist in a
   627→   627→  // browser — so after boot we override it with a small Python shim that
   628→   628→  // calls window.prompt() instead. Cancelling the prompt raises EOFError
   629→   629→  // (shown as a red traceback), which is exactly what real Python does on
   630→   630→  // an EOF. The override is wrapped in try/except-equivalent so a failure
   631→   631→  // here can NEVER block the boot sequence.
   632→   632→  var INPUT_OVERRIDE_PY = `
   633→   633→import builtins, js
   634→   634→
   635→   635→def _input(prompt=""):
   636→   636→    reply = js.window.prompt(str(prompt))
   637→   637→    if reply is None:
   638→   638→        raise EOFError("input() cancelled")
   639→   639→    return str(reply)
   640→   640→
   641→   641→builtins.input = _input
   642→   642→`;
   643→   643→
   644→   644→  function installInputOverride() {
   645→   645→    try {
   646→   646→      pyodide.runPython(INPUT_OVERRIDE_PY);
   647→   647→    } catch (err) {
   648→   648→      // input() simply keeps its default behavior if this fails.
   649→   649→      console.warn("[Code Lab] Could not install input() override:", err);
   650→   650→    }
   651→   651→  }
   652→   652→
   653→   653→  // ================= 6. RUN / CLEAR / STATUS =================
   654→   654→  // setStatus(state) drives the status pill through its states:
   655→   655→  //   booting → "Booting Python runtime…" (spinner + amber dot)
   656→   656→  //   ready   → "Python 3 · ready"        (green dot)
   657→   657→  //   running → "Running…"                (blue pulsing dot)
   658→   658→  //   error   → "Python failed to load"   (red dot, CDN failure only)
   659→   659→  function setStatus(state) {
   660→   660→    statusPill.classList.remove("booting", "ready", "running", "error");
   661→   661→    statusPill.classList.add(state);
   662→   662→    statusText.textContent = {
   663→   663→      booting: "Booting Python runtime…",
   664→   664→      ready:   "Python 3 · ready",
   665→   665→      running: "Running…",
   666→   666→      error:   "Python failed to load"
   667→   667→    }[state];
   668→   668→  }
   669→   669→
   670→   670→  // runCode() executes the editor's Python code:
   671→   671→  //   • ignored when the runtime is not ready or a run is already active
   672→   672→  //   • clears the console, prints a muted "▸ run started"
   673→   673→  //   • awaits pyodide.runPythonAsync(code); any Python traceback arrives
   674→   674→  //     as err.message and is printed in RED without crashing the page
   675→   675→  //   • a `finally` block always restores the UI state (muted "▸ run
   676→   676→  //     finished", button re-enabled, pill back to "Python 3 · ready")
   677→   677→  function runCode() {
   678→   678→    if (!pyReady || pyRunning || !pyodide) { return; } // do nothing if not ready/busy
   679→   679→    pyRunning = true;
   680→   680→    runBtn.disabled = true;
   681→   681→    setStatus("running");
   682→   682→
   683→   683→    clearOutput();                          // fresh console for each run
   684→   684→    appendOutput("\u25B8 run started\n", "muted");
   685→   685→
   686→   686→    var code = editor.getValue();
   687→   687→    pyodide.runPythonAsync(code)
   688→   688→      .catch(function (err) {
   689→   689→        appendOutput((err && err.message ? err.message : String(err)) + "\n", "err");
   690→   690→      })
   691→   691→      .then(function () {
   692→   692→        // `finally` equivalent: always runs, success or failure.
   693→   693→        pyRunning = false;
   694→   694→        appendOutput("\u25B8 run finished\n", "muted");
   695→   695→        if (pyReady) {
   696→   696→          runBtn.disabled = false;
   697→   697→          setStatus("ready");
   698→   698→        }
   699→   699→      });
   700→   700→  }
   701→   701→
   702→   702→  // Wire up the toolbar buttons.
   703→   703→  runBtn.addEventListener("click", runCode);
   704→   704→  clearBtn.addEventListener("click", clearOutput);
   705→   705→
   706→   706→  // ================= 7. CROSS-FRAME API (set-code) =================
   707→   707→  // When embedded in an iframe, the host notebook site can push code into
   708→   708→  // the editor by posting a message:
   709→   709→  //     iframe.contentWindow.postMessage({ type: "set-code", code: "..." }, "*")
   710→   710→  // This listener is harmless when the page is opened standalone (no
   711→   711→  // messages ever arrive).
   712→   712→  window.addEventListener("message", function (event) {
   713→   713→    if (event.data && event.data.type === "set-code") {
   714→   714→      editor.setValue(String(event.data.code ?? ""));
   715→   715→    }
   716→   716→  });
   717→   717→
   718→   718→  // Once Pyodide is ready, tell the host page we can accept code / runs.
   719→   719→  // When standalone, window.parent is this same window — posting to
   720→   720→  // ourselves is harmless (the listener above only reacts to "set-code").
   721→   721→  function notifyHostReady() {
   722→   722→    try {
   723→   723→      window.parent.postMessage({ type: "pyodide-ready" }, "*");
   724→   724→    } catch (err) { /* ignore — standalone page or restricted sandbox */ }
   725→   725→  }
   726→   726→
   727→   727→  /* ------------------------------------------------------------------
   728→   728→     Kick off the boot sequence when the page has fully loaded.
   729→   729→     ------------------------------------------------------------------ */
   730→   730→  if (document.readyState === "complete") {
   731→   731→    bootPyodide();
   732→   732→  } else {
   733→   733→    window.addEventListener("load", bootPyodide);
   734→   734→  }
   735→   735→})();
